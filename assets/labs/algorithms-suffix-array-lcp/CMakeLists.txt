cmake_minimum_required(VERSION 3.20)
project(algorithms_suffix_array_lcp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)

set(CMAKE_CXX_EXTENSIONS OFF)

enable_testing()
