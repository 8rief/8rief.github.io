cmake_minimum_required(VERSION 3.20)
project(algorithms_randomized_quickselect LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(quickselect_demo src/main.cpp)
target_compile_options(quickselect_demo PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME quickselect_demo_tests COMMAND quickselect_demo --test)
