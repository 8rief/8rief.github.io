cmake_minimum_required(VERSION 3.20)
project(algorithms_ntt_convolution LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(ntt_demo src/main.cpp)
target_compile_options(ntt_demo PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME ntt_demo_tests COMMAND ntt_demo --test)
