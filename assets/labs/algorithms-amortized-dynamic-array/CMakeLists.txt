cmake_minimum_required(VERSION 3.20)
project(algorithms_amortized_dynamic_array LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(amortized_array_demo src/main.cpp)
target_compile_options(amortized_array_demo PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME amortized_array_demo_tests COMMAND amortized_array_demo --test)
