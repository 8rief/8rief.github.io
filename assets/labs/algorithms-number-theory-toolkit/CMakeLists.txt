cmake_minimum_required(VERSION 3.20)
project(algorithms_number_theory_toolkit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(number_theory_demo src/main.cpp)
target_compile_options(number_theory_demo PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME number_theory_demo_tests COMMAND number_theory_demo --test)
