cmake_minimum_required(VERSION 3.20)
project(algorithms_binary_lifting_lca LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(lca_demo src/main.cpp)
target_compile_options(lca_demo PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME lca_demo_tests COMMAND lca_demo --test)
